// ================= 屏幕引脚强制配置区 =================
#define ST7789_DRIVER 
#define TFT_WIDTH  240
#define TFT_HEIGHT 320
#define TFT_MOSI 23
#define TFT_SCLK 18
#define TFT_CS   15
#define TFT_DC   27
#define TFT_RST   4
#define TFT_MISO -1 
#define SPI_FREQUENCY  10000000
// =======================================================

#include <WiFi.h>
#include <PubSubClient.h>
#include <TFT_eSPI.h>
#include <U8g2_for_TFT_eSPI.h>

// ================= 配置区 =================
const char* WIFI_SSID     = "1";           
const char* WIFI_PASSWORD = "你的WiFi密码";   
const char* MQTT_BROKER   = "broker.emqx.io";

// 【手语通道】接收电脑发来的文字
const char* TOPIC_SIGN_TEXT  = "aiot/sign/gesture/demo"; 
// 【语音通道】ESP32发音频给电脑，电脑发文字回ESP32
const char* TOPIC_AUDIO_SEND = "aiot/voice/audio"; 
const char* TOPIC_AUDIO_TEXT = "aiot/voice/text"; 

#define PIN_MIC_ADC 34
// ==========================================

WiFiClient espClient;
PubSubClient mqtt(espClient);
TFT_eSPI tft = TFT_eSPI();
U8g2_for_TFT_eSPI u8f;

// 录音参数
#define SAMPLE_RATE 16000
#define RECORD_SECONDS 3 
#define ADC_SAMPLE_COUNT (SAMPLE_RATE * RECORD_SECONDS)
int16_t *pcmBuffer = nullptr;
volatile bool adcRecording = false;
volatile uint32_t adcSampleIndex = 0;
hw_timer_t *adcTimer = nullptr;

unsigned long lastWifiRetry = 0;
unsigned long lastMqttRetry = 0;
const unsigned long RETRY_INTERVAL = 5000;

// ================= 屏幕显示函数 =================
void showSignText(const String &text) {
  tft.fillRect(0, 30, 320, 90, TFT_BLACK);
  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_YELLOW);
  u8f.drawUTF8(20, 90, text.c_str());
}

void showVoiceText(const String &text) {
  tft.fillRect(0, 151, 320, 89, TFT_BLACK);
  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_CYAN);
  u8f.drawUTF8(20, 200, text.c_str());
}

// ================= MQTT 回调 =================
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String text = "";
  for (unsigned int i = 0; i < length; i++) text += (char)payload[i];
  text.trim();
  if(text.length() == 0) return;

  // 根据收到的主题，各自刷新对应的区域
  if (String(topic) == TOPIC_SIGN_TEXT) {
    showSignText(text);
  } else if (String(topic) == TOPIC_AUDIO_TEXT) {
    showVoiceText(text);
  }
}

// ================= 定时器音频采样 =================
void IRAM_ATTR onAdcTimer() {
  if (!adcRecording) return;
  if (adcSampleIndex >= ADC_SAMPLE_COUNT) {
    adcRecording = false;
    return;
  }
  int raw = analogRead(PIN_MIC_ADC);
  int16_t sample = (int16_t)((raw - 2048) << 4); 
  pcmBuffer[adcSampleIndex++] = sample;
}

void initTimer() {
  adcTimer = timerBegin(SAMPLE_RATE);                
  timerAttachInterrupt(adcTimer, &onAdcTimer);       
  timerAlarm(adcTimer, 1, true, 0);                 
}

// ================= 网络重连 =================
void ensureWiFi() {
  if (WiFi.status() == WL_CONNECTED) return;
  unsigned long now = millis();
  if (now - lastWifiRetry < RETRY_INTERVAL) return;
  lastWifiRetry = now;
  WiFi.disconnect();
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
}

void ensureMQTT() {
  if (WiFi.status() != WL_CONNECTED) return;
  if (mqtt.connected()) return;
  unsigned long now = millis();
  if (now - lastMqttRetry < RETRY_INTERVAL) return;
  lastMqttRetry = now;
  if (mqtt.connect("ESP32_Final_BiChannel")) {
    mqtt.subscribe(TOPIC_SIGN_TEXT);  
    mqtt.subscribe(TOPIC_AUDIO_TEXT); 
  }
}

// ================= Setup =================
void setup() {
  Serial.begin(115200);

  tft.init();
  tft.setRotation(1);
  tft.fillScreen(TFT_BLACK);
  u8f.begin(tft);
  u8f.setFontMode(1);

  tft.fillRect(0, 0, 320, 30, TFT_NAVY);
  tft.setTextColor(TFT_WHITE);
  tft.setTextSize(2);
  tft.setCursor(10, 6);
  tft.print("Sign Translation");

  tft.drawFastHLine(0, 120, 320, TFT_LIGHTGREY);
  tft.fillRect(0, 121, 320, 30, TFT_DARKGREEN);
  tft.setCursor(10, 126);
  tft.print("Voice Transcription");

  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_DARKGREY);
  u8f.drawUTF8(20, 90, "等待手语...");
  u8f.drawUTF8(20, 200, "等待语音...");

  pcmBuffer = (int16_t *)malloc(ADC_SAMPLE_COUNT * sizeof(int16_t));
  initTimer();

  WiFi.mode(WIFI_STA);
  ensureWiFi();
  
  mqtt.setBufferSize(100 * 1024); 
  mqtt.setServer(MQTT_BROKER, 1883);
  mqtt.setCallback(mqttCallback);
  
  adcSampleIndex = 0;
  adcRecording = true; // 上电即开启麦克风后台录音
}

// ================= Loop =================
void loop() {
  ensureWiFi();
  ensureMQTT();
  mqtt.loop();

  // 【核心逻辑】：绝对静默，只发数据，不改变屏幕
  if (!adcRecording) {
    if (mqtt.connected()) {
      mqtt.publish(TOPIC_AUDIO_SEND, (const uint8_t*)pcmBuffer, ADC_SAMPLE_COUNT * sizeof(int16_t));
    }
    delay(1000); 
    adcSampleIndex = 0;
    adcRecording = true; 
  }

  delay(10);
}// ================= 屏幕引脚强制配置区 =================

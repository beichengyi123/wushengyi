// ================= 无声有译 · ESP32 终端（语音修复版 v2） =================
// 屏幕逻辑与 MQTT 通道与原版完全一致，仅重写音频采集：
//   旧版: 16kHz 硬件定时器中断里调 analogRead —— 中断耗时超过采样间隔，必丢样
//   新版: loop 里用 adc1_get_raw + micros 定时，8kHz 采样（语音识别够用），
//         分块录音不阻塞网络，录完打印 swing 便于诊断麦克风
// 屏幕配置由 User_Setup.h 提供（ILI9341, CS=15 DC=27 MOSI=23 SCK=18 BL=32）

#include <WiFi.h>
#include <PubSubClient.h>
#include <driver/adc.h>
#include <TFT_eSPI.h>
#include <U8g2_for_TFT_eSPI.h>

// ================= 配置区 =================
const char* WIFI_SSID     = "1";
const char* WIFI_PASSWORD = "你的WiFi密码";
const char* MQTT_BROKER   = "broker.emqx.io";

const char* TOPIC_SIGN_TEXT  = "aiot/sign/gesture/demo";
const char* TOPIC_AUDIO_SEND = "aiot/voice/audio";
const char* TOPIC_AUDIO_TEXT = "aiot/voice/text";

#define PIN_MIC_ADC 34            // 模拟麦克风 OUT
#define MIC_CHANNEL ADC1_CHANNEL_6 // GPIO34 对应 ADC1_CH6
// ==========================================

WiFiClient espClient;
PubSubClient mqtt(espClient);
TFT_eSPI tft = TFT_eSPI();
U8g2_for_TFT_eSPI u8f;

// ---- 录音参数（8kHz x 3秒 = 24000 点，静态分配 48KB）----
#define SAMPLE_RATE     8000
#define RECORD_SECONDS  3
#define TOTAL_SAMPLES   (SAMPLE_RATE * RECORD_SECONDS)
#define CHUNK_SAMPLES   800        // 每次循环录 0.1 秒，期间让网络喘气
static int16_t pcmBuffer[TOTAL_SAMPLES];
static uint32_t sampleIdx = 0;
static bool recording = true;
static uint32_t nextTickUs = 0;
static uint32_t publishCount = 0;

unsigned long lastWifiRetry = 0;
unsigned long lastMqttRetry = 0;
const unsigned long RETRY_INTERVAL = 5000;

// ================= 屏幕显示函数 =================
void showSignText(const String &text) {
  tft.fillRect(0, 30, 320, 90, TFT_BLACK);
  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_YELLOW);
  u8f.drawUTF8(20, 90, text.c_str());
}

void showVoiceText(const String &text) {
  tft.fillRect(0, 151, 320, 89, TFT_BLACK);
  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_CYAN);
  u8f.drawUTF8(20, 200, text.c_str());
}

// ================= MQTT 回调 =================
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String text = "";
  for (unsigned int i = 0; i < length; i++) text += (char)payload[i];
  text.trim();
  if (text.length() == 0) return;

  if (String(topic) == TOPIC_SIGN_TEXT) {
    showSignText(text);
  } else if (String(topic) == TOPIC_AUDIO_TEXT) {
    showVoiceText(text);
  }
}

// ================= 分块录音（8kHz，micros 定时）=================
void recordChunk() {
  uint32_t now = micros();
  for (int i = 0; i < CHUNK_SAMPLES; i++) {
    if (sampleIdx >= TOTAL_SAMPLES) break;
    // 等到下一个采样时刻
    while ((int32_t)(now - nextTickUs) < 0) { now = micros(); }
    int raw = adc1_get_raw(MIC_CHANNEL);
    pcmBuffer[sampleIdx++] = (int16_t)((raw - 2048) << 4);
    nextTickUs += (1000000UL / SAMPLE_RATE);
  }
}

// 录完一轮，算波动并打印（麦克风诊断核心数据）
void reportAndPublish() {
  int mn = 32767, mx = -32768;
  for (int i = 0; i < TOTAL_SAMPLES; i++) {
    if (pcmBuffer[i] < mn) mn = pcmBuffer[i];
    if (pcmBuffer[i] > mx) mx = pcmBuffer[i];
  }
  int swing = mx - mn;
  publishCount++;
  Serial.printf("[录音 #%d 完成] min=%d max=%d swing=%d %s\n",
                publishCount, mn, mx, swing,
                swing > 3000 ? "(有声音输入)" : "(几乎无声!)");

  if (mqtt.connected()) {
    bool ok = mqtt.publish(TOPIC_AUDIO_SEND, (const uint8_t*)pcmBuffer,
                           TOTAL_SAMPLES * sizeof(int16_t));
    Serial.printf("[MQTT] 音频上传 %s (%d 字节)\n",
                  ok ? "成功" : "失败!", TOTAL_SAMPLES * 2);
  } else {
    Serial.println("[MQTT] 未连接，本轮不上传");
  }
}

// ================= 网络重连 =================
void ensureWiFi() {
  if (WiFi.status() == WL_CONNECTED) return;
  unsigned long now = millis();
  if (now - lastWifiRetry < RETRY_INTERVAL) return;
  lastWifiRetry = now;
  WiFi.disconnect();
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
}

void ensureMQTT() {
  if (WiFi.status() != WL_CONNECTED) return;
  if (mqtt.connected()) return;
  unsigned long now = millis();
  if (now - lastMqttRetry < RETRY_INTERVAL) return;
  lastMqttRetry = now;
  Serial.println("尝试连接 MQTT...");
  if (mqtt.connect("ESP32_Final_BiChannel")) {
    Serial.println("MQTT 成功");
    mqtt.subscribe(TOPIC_SIGN_TEXT);
    mqtt.subscribe(TOPIC_AUDIO_TEXT);
  }
}

// ================= Setup =================
void setup() {
  Serial.begin(115200);
  delay(200);

  // --- 麦克风 ADC 初始化 ---
  adc1_config_width(ADC_WIDTH_BIT_12);
  adc1_config_channel_atten(MIC_CHANNEL, ADC_ATTEN_DB_11); // 0~3.3V 全量程
  Serial.println("[麦克风] ADC 就绪 (GPIO34)");

  // --- 屏幕 ---
  tft.init();
  tft.setRotation(1);
  tft.fillScreen(TFT_BLACK);
  u8f.begin(tft);
  u8f.setFontMode(1);

  tft.fillRect(0, 0, 320, 30, TFT_NAVY);
  tft.setTextColor(TFT_WHITE);
  tft.setTextSize(2);
  tft.setCursor(10, 6);
  tft.print("Sign Translation");

  tft.drawFastHLine(0, 120, 320, TFT_LIGHTGREY);
  tft.fillRect(0, 121, 320, 30, TFT_DARKGREEN);
  tft.setCursor(10, 126);
  tft.print("Voice Transcription");

  u8f.setFont(u8g2_font_wqy16_t_gb2312);
  u8f.setForegroundColor(TFT_DARKGREY);
  u8f.drawUTF8(20, 90, "等待手语...");
  u8f.drawUTF8(20, 200, "等待语音...");

  // --- 网络 ---
  WiFi.mode(WIFI_STA);
  ensureWiFi();

  mqtt.setBufferSize(65000);   // 注意: 库的参数是16位整数(最大65535), 写64*1024=65536会溢出归零!
  mqtt.setServer(MQTT_BROKER, 1883);
  mqtt.setCallback(mqttCallback);

  nextTickUs = micros();
}

// ================= Loop =================
void loop() {
  ensureWiFi();
  ensureMQTT();
  mqtt.loop();

  if (recording) {
    recordChunk();
    if (sampleIdx >= TOTAL_SAMPLES) {
      recording = false;
      reportAndPublish();          // 录完：打印诊断 + 上传
      delay(1000);                 // 休息 1 秒再录下一轮
      sampleIdx = 0;
      nextTickUs = micros();
      recording = true;
    }
  }

  delay(2);
}
